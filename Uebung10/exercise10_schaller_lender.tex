\section*{10.1: Density matrix of a single qubit}
\subsection*{(a)}
A general $2×2$ density matrix $\rho$ for a qubit can be written as:
\begin{align*}
    \rho=
    \begin{pmatrix}
      \rho_{11}&\rho_{12}\\ 
      \rho_{21}&\rho_{22}
    \end{pmatrix}
\end{align*}
It must be hermitian$(\rho_{12}=\rho_{21}^*)$ and have a unit trace $(\text{Tr}(ρ)=\rho_{11}+\rho_{22}=1)$. We can parametrize it using a real number a and a complex number c:
\begin{align}
    \rho=
    \begin{pmatrix}
      a&c\\ 
      c^*&1-a
    \end{pmatrix}
\end{align}

The expectation value of an operator M is given by $⟨M⟩=\text{Tr}(ρM)$. We use the given expectation values to find the unknown variables:

with $⟨A⟩=2$:
\begin{align}
    A&=
    \begin{pmatrix}
      3&0\\ 
      0&1
    \end{pmatrix}\\[0.2cm]
    \text{Tr}(ρA)&=\text{Tr}
    \begin{pmatrix}
    3a&c\\
    3c^*&1-a
    \end{pmatrix}
    =3a+(1−a)=2a+1\\[0.2cm]
    2&=2a+1 \Rightarrow a=\frac{1}{2} 
\end{align}
Thus, the diagonal entries are $\rho_{11}=\frac{1}{2}$ and $\rho_{22}=1−\frac{1}{2}=\frac{1}{2}$.

with $⟨B⟩=\frac{1}{2}$:
\begin{align}
    B&=
    \begin{pmatrix}
      1&1\\ 
      1&-1
    \end{pmatrix}\\[0.2cm]
    \text{Tr}(ρB)&=\text{Tr}
    \begin{pmatrix}
    \frac{1}{2}+c&\frac{1}{2}-c\\
    c^*+\frac{1}{2}&c^*-\frac{1}{2}
    \end{pmatrix}
    =\left(\frac{1}{2}+c\right)+\left(c^*-\frac{1}{2}\right)=c+c^*=2\text{Re}(c)\\[0.2cm]
    0&=4\text{Im}(c) \Rightarrow \text{Im}(c)=0 
\end{align}

with $⟨C⟩$=0:
\begin{align}
    C&=
    \begin{pmatrix}
      0&2i\\ 
      -2i&0
    \end{pmatrix}\\[0.2cm]
    \text{Tr}(ρC)&=\text{Tr}
    \begin{pmatrix}
    -2ic&i\\
    -i&2ic^*
    \end{pmatrix}
    =\text{Tr}(ρC)=−2ic+2ic^∗=2i(c^∗−c)=2i(−2i\text{Im}(c))=4\text{Im}(c)\\[0.2cm]
    \frac{1}{2}&=2\text{Re}(c) \Rightarrow \text{Re}(c)=\frac{1}{2} 
\end{align}

So the density matrix is:
\begin{align}
    \rho=
    \begin{pmatrix}
        \frac{1}{2}&\frac{1}{4}\\
        \frac{1}{4}&\frac{1}{2}
    \end{pmatrix}
    =
    \frac{1}{4}
    \begin{pmatrix}
        2&1\\
        1&2
    \end{pmatrix}
\end{align}

    \frac{1}{16}

    \end{pmatrix}

\end{align}

\subsection*{(c)}
The Pauli-z operator is $\sigma_z=\begin{pmatrix}
1&0\\
0&-1
\end{pmatrix}$.
The eigenvalue +1 corresponds to the basis state $∣0⟩=\begin{pmatrix}
1\\
0
\end{pmatrix}$.
The probability of obtaining an outcome corresponding to a projection operator $P=∣0⟩⟨0∣$ is given by:
\begin{align}
    P(+1)=\langle0|\rho|0\rangle=\rho_{11}=\frac{1}{2}
\end{align}
Therefore the probability to get +1 is 50\%.

    \langle\sigma_x\rangle&=\text{Tr}(\rho\sigma_x)= \frac{1}{4}+\frac{1}{4}=\frac{1}{2}\\
    \langle\sigma_y\rangle&=\text{Tr}(\rho\sigma_y)= -i\frac{1}{4}+i\frac{1}{4}= 0\\
    \langle\sigma_z\rangle&=\text{Tr}(\rho\sigma_z)= \frac{1}{2}-\frac{1}{2}= 0
\end{align}


\section*{10.2: Criterion for the equivalence of ensembles}
\subsection*{(a)}
\begin{align}
    ρ_{\psi}=\sum_{i=1}^Np_i∣ψ_i⟩⟨ψ_i∣\\
    ρ_{k}=\sum_{i=1}^K\lambda_k∣k⟩⟨k∣
\end{align}
Look at a state $|\phi\rangle$, that is orthogonal to $\rho_k$ (meaning $\rho_k|\phi\rangle=0$), therefore $\rho_\psi|\phi\rangle=0$, now we look closer at the expectation value $\langle\rho\rangle$:
\begin{align}
    \langle\phi|\rho|\phi\rangle=\sum_{i=1}^Np_i\langle\phi|\psi_i\rangle\langle\psi_i|\phi\rangle=\sum_{i=1}^Np_i|\langle\phi|\psi_i\rangle|^2=0
    \intertext{Because the probabilities $p_i$ are strictly positive $(p_i>0)$ and the squared magnitudes $|\langle\phi|\psi_i\rangle|^2$ are non-negative, the only way the sum can equal zero is if every individual term is zero:}
    |\langle\phi|\psi_i\rangle|^2=0 \Rightarrow \langle\phi|\psi_i\rangle=0 \hspace{1cm}\text{for all i}
\end{align}

Since any vector $∣ϕ⟩$ orthogonal to $\rho_k$ (and therefore to span$(\{∣k⟩\})$) is also orthogonal to every $∣ψ_i⟩$, it follows that the vectors $∣ψ_i⟩$ must lie entirely within the subspace spanned by the eigenvectors.

Thus, each $∣ψ_i⟩$ can be written as a linear combination of the eigenvectors $∣k⟩$:
\begin{align}
    |\psi_i\rangle\in\text{span}(\{|k\rangle\})
\end{align}
