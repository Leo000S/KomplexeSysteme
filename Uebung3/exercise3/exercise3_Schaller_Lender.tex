\section*{Task 3.1: Central Limit Theorem}
\subsection*{(a)}
\begin{align}
    \text{for N=1:}\hspace{0.5cm} p_{Z_1}(z)&=\frac{z^{1-1}}{(1-1)!}e^{-z}=\frac{1}{1}e^{-z}=e^{-z}=p_X(z)\\
    \text{assume for k$\ge$ 1:}\hspace{0.5cm} p_{Z_k}(z)&=\frac{z^{k-1}}{(k-1)!}e^{-z}\\
    \text{for k+1:}\hspace{0.5cm}  p_{Z_{k+1}}(z)&= (p_{Z_k}\ast p_X)(z)=\int_{-\infty}^{\infty}p_{Z_k}(x)\ p_X(z-x)\ dx
\end{align}
p(x) is only non zero for $x\ge0$, therefore p(z-x) is only non zero for $z-x\ge0\rightarrow z\ge x$. With this information we can plug in $p_{Z_k}(x)$ and rewrite the integral:
\begin{align}
     p_{Z_{k+1}}(z)&=\int_{-\infty}^{\infty}p_{Z_k}(x)\ p_X(z-x)\ dx\\
     &=\int_0^z \frac{x^{k-1}}{(k-1)!}e^{-x} \cdot e^{-(z-x)}\ dx\\
     &=\frac{e^{-z}}{(k-1)!}\int_0^zx^{k-1} dx \\
     &=\frac{e^{-z}}{(k-1)!}\frac{z^{k}}{k}=\frac{z^{k}}{k!}e^{-z}
\end{align}
Now the proof via induction is complete.

\section*{Task 3.2: Vector notation and Liouville operator}
\subsection*{(a)}

\begin{align}
    \dot{P}_c(t)=\sum_{c'}P_{c'}(t)\ \omega_{c'\rightarrow c}-P_{c}(t)\sum_{c''}\ \omega_{c\rightarrow c''}
\end{align}
to write the whole term as a sum over $c'$, we use the Kronecker delta $\delta_{c',c}$. With $\delta_{c',c}P_c=\delta_{c',c}P_{c'}$ we get:
\begin{align}
    \dot{P}_c(t)&=\sum_{c'}P_{c'}(t)\ \omega_{c'\rightarrow c}- \sum_{c'}\delta_{c',c}\Big(\sum_{c''}\ \omega_{c\rightarrow c''}\Big)P_{c'}(t)\\
    &=\sum_{c'}\Big(\omega_{c'\rightarrow c}- \delta_{c',c}\sum_{c''}\omega_{c\rightarrow c''}\Big)P_{c'}(t)
\end{align}
Now we expand at the Operator representation of the probability vector:
\begin{align}
    \ket{{P}}&=\sum_{c'} P_{c'}(t)\ket{c'}\\
    \Rightarrow \small\braket*{c}{\dot{P}}&=-\mel{c}{\mathcal{L}}{P}=-\sum_{c'}\mel{c}{\mathcal{L}}{c'}P_{c'}(t)\\
    &=\dot{P}_{c}(t)
\end{align}
A comparison of both results shows:
\begin{align}
    -\mel{c}{\mathcal{L}}{c'}=\omega_{c'\rightarrow c}- \delta_{c',c}\sum_{c''}\omega_{c\rightarrow c''}
\end{align}
no we relabel the indices ($c\rightarrow c', c'\rightarrow c$) and get the expression we should prove:
\begin{align}
    \mel{c}{\mathcal{L}}{c'}=-\omega_{c\rightarrow c'}- \delta_{c,c'}\sum_{c''}\omega_{c\rightarrow c''}
\end{align}
