\section*{Task 5.1: Tensor product}
\subsection*{(a)}

to prove:
\begin{align}
    e^{\mathbf{A}\otimes \mathbb{1}+\mathbb{1}\otimes \mathbf{B}}=e^\mathbf{A}\otimes e^\mathbf{B}
\end{align}
first we show that $\mathbf{X}=\mathbf{A}\otimes \mathbb{1}$ and $\mathbf{Y}=\mathbb{1}\otimes \mathbf{B}$ commute via the relation 
\begin{align}
(\mathbf{A}\otimes \mathbf{B})(\mathbf{C}\otimes \mathbf{D})=(\mathbf{AC})\otimes (\mathbf{BD})
\label{relation}
\end{align}
which emerges directly from the definition of the Kronecker product.
\begin{align}
   [\mathbf{X},\mathbf{Y}]&=(\mathbf{A}\otimes \mathbb{1})(\mathbb{1}\otimes \mathbf{B})-(\mathbb{1}\otimes \mathbf{B})(\mathbf{A}\otimes \mathbb{1})\\
   &=\mathbf{A}\mathbb{1}\otimes\mathbb{1}B-\mathbb{1}\mathbf{A}\otimes \mathbf{B}\mathbb{1}= \mathbf{A}\otimes \mathbf{B} -\mathbf{A}\otimes \mathbf{B}=0
\end{align}
now we can use:
\begin{align}
    e^{\mathbf{A}\otimes \mathbb{1}+\mathbb{1}\otimes \mathbf{B}}=e^{\mathbf{X}+\mathbf{Y}}=e^\mathbf{X}e^\mathbf{Y}= e^{\mathbf{A}\otimes \mathbb{1}}e^{\mathbb{1}\otimes \mathbf{B}}
\end{align}
the general matrix exponential is 
\begin{align}
    e^\mathbf{X}&=\sum^\infty_{k=0}\frac{\mathbf{\mathbf{X}}^k}{k!}\\
    \Rightarrow     e^{\mathbf{A}\otimes \mathbb{1}}&=\sum^\infty_{k=0}\frac{(\mathbf{A}\otimes\mathbb{1})^k}{k!}
\end{align}
applying Eq.\ref{relation} repeatedly on $(\mathbf{A}\otimes\mathbb{1})^k$:
\begin{align}
(\mathbf{A}\otimes\mathbb{1})^k&=(\mathbf{A}\otimes\mathbb{1})(\mathbf{A}\otimes\mathbb{1})(\mathbf{A}\otimes\mathbb{1})^{k-2}=\\
    &=(\mathbf{A}^2\otimes\mathbb{1}^2)(\mathbf{A}\otimes\mathbb{1})(\mathbf{A}\otimes\mathbb{1})(\mathbf{A}\otimes\mathbb{1})^{k-4}\\
    &=(\mathbf{A}^2\otimes\mathbb{1}^2)(\mathbf{A}^2\otimes\mathbb{1^2})(\mathbf{A}\otimes\mathbb{1})^{k-4}\\
    &=(\mathbf{A}^4\otimes\mathbb{1}^4)(\mathbf{A}\otimes\mathbb{1})^{k-4}= ...\\
    &=\mathbf{A}^k\otimes\mathbb{1}^k
\end{align}
now we can write the exponential as:
\begin{align}
    e^{\mathbf{A}\otimes \mathbb{1}}&= \sum^\infty_{k=0}\frac{(\mathbf{A}\otimes\mathbb{1})^k}{k!}=\sum^\infty_{k=0}\frac{\mathbf{A}^k\otimes\mathbb{1}^k}{k!}\\
    \intertext{since $\mathbb{1}^k=\mathbb{1}$:}e^{\mathbf{A}\otimes \mathbb{1}}&=\sum^\infty_{k=0}\frac{\mathbf{A}^k}{k!}\otimes\mathbb{1}=e^\mathbf{A}\otimes\mathbb{1}
\end{align}
following the same arguments we see $e^{\mathbb{1}\otimes \mathbf{B}}=\mathbb{1}\otimes e^\mathbf{B}$ and thus we showed
\begin{align}
    e^{A\otimes \mathbb{1}+\mathbb{1}\otimes \mathbf{B}}&=e^{A\otimes \mathbb{1}}e^{\mathbb{1}\otimes \mathbf{B}}= (e^\mathbf{A}\otimes\mathbb{1})(\mathbb{1}\otimes e^\mathbf{B})\\
    &=(e^\mathbf{A}\mathbb{1})\otimes(\mathbb{1}e^\mathbf{B})=e^\mathbf{A}\otimes e^\mathbf{B}
\end{align}

\subsection*{(b)}
consider the state $\ket{a}\otimes\ket{b}$:
\begin{align}
    (\mathbf{A} \otimes \mathbf{B})(|a\rangle \otimes |b\rangle) &= (\mathbf{A}|a\rangle) \otimes (\mathbf{B}|b\rangle)\\
   \intertext{substitute given eigenvalues $\mathbf{A}|a\rangle = a|a\rangle$ and $\mathbf{B}|b\rangle = b|b\rangle$:} &=(a|a\rangle) \otimes (b|b\rangle) = ab(|a\rangle \otimes |b\rangle) 
\end{align}
So the eigenvalues are $ab$ with the eigenvectors $(|a\rangle \otimes |b\rangle)$.

\subsection*{(c)}
the standard Pauli Commutator is
\begin{align}
    [\sigma_i, \sigma_j] = 2i \epsilon_{ijk} \sigma_k
\end{align}
for the Tensor Product:
\begin{align}
    [\sigma_i \otimes \sigma_i, \sigma_j \otimes \sigma_j] &= (\sigma_i \otimes \sigma_i)(\sigma_j \otimes \sigma_j) - (\sigma_j \otimes \sigma_j)(\sigma_i \otimes \sigma_i)\\
    &=(\sigma_i \sigma_j \otimes \sigma_i \sigma_j) - (\sigma_j \sigma_i \otimes \sigma_j \sigma_i)
\end{align}
with $\sigma_i \sigma_j=\delta_{i,j}\mathbb{1}+i \epsilon_{ijk} \sigma_k$ we have to look at two cases.

case 1: $i=j \Rightarrow \sigma_i \sigma_j=\mathbb{1}$. The Tensor Product Commutator is then:
\begin{align}
    (\sigma_i \sigma_j \otimes \sigma_i \sigma_j) - (\sigma_j \sigma_i \otimes \sigma_j \sigma_i)&= (\mathbb{1}\otimes\mathbb{1})-(\mathbb{1}\otimes\mathbb{1})=0
\intertext{case 2: $i\neq j \Rightarrow \sigma_i \sigma_j=i \epsilon_{ijk} \sigma_k$. The Tensor Product Commutator is then:}
    (\sigma_i \sigma_j \otimes \sigma_i \sigma_j) - (\sigma_j \sigma_i \otimes \sigma_j \sigma_i)&= (i \sigma_k\otimes i\sigma_k)-(-i \sigma_k\otimes-i\sigma_k)\\
    &=-(\sigma_k\otimes\sigma_k)+(\sigma_k\otimes\sigma_k)= 0
\intertext{so for every $i,j$:}
    [\sigma_i \otimes \sigma_i, \sigma_j \otimes \sigma_j]&=0
\end{align}

\begin{align}
     e^{\phi \sigma_i \otimes \sigma_j}&= \sum_{n=0}^{\infty} \frac{(\phi \sigma_i \otimes \sigma_j)^n}{n!}\\
     \intertext{with}
     (\sigma_i \otimes \sigma_j)^2&=(\sigma_i \otimes \sigma_j)(\sigma_i \otimes \sigma_j)= (\sigma_i^2 \otimes \sigma_j^2)= \mathbb{1}\otimes\mathbb{1}=\mathbb{I}\\
     \intertext{we can write}
     \sum_{n=0}^{\infty} \frac{(\phi \sigma_i \otimes \sigma_j)^n}{n!} &= \mathbb{I}\sum_{k=0}^{\infty} \frac{\phi^{2k}}{(2k)!}+ \sigma_i \otimes \sigma_j\sum_{k=0}^{\infty} \frac{\phi^{2k+1}}{(2k+1)!}\\
     \intertext{the series on the right side correspond to the hyperbolic functions.}
     \Rightarrow  e^{\phi \sigma_i \otimes \sigma_j}&= \mathbb{I} \cosh(\phi) + (\sigma_i \otimes \sigma_j) \sinh(\phi)
\end{align}
