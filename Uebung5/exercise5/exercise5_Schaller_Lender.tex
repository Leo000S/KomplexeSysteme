\section*{Task 5.1: Tensor product}
\subsection*{(a)}

to prove:
\begin{align}
    e^{\mathbf{A}\otimes \mathbf{1}+\mathbf{1}\otimes \mathbf{B}}=e^\mathbf{A}\otimes e^\mathbf{B}
\end{align}
first we show that $\mathbf{X}=\mathbf{A}\otimes \mathbf{1}$ and $\mathbf{Y}=\mathbf{1}\otimes \mathbf{B}$ commute via the relation 
\begin{align}
(\mathbf{A}\otimes \mathbf{B})(\mathbf{C}\otimes \mathbf{D})=(\mathbf{AC})\otimes (\mathbf{BD})
\label{relation}
\end{align}
which emerges directly from the definition of the Kronecker product.
\begin{align}
   [\mathbf{X},\mathbf{Y}]&=(\mathbf{A}\otimes \mathbf{1})(\mathbf{1}\otimes \mathbf{B})-(\mathbf{1}\otimes \mathbf{B})(\mathbf{A}\otimes \mathbf{1})\\
   &=\mathbf{A}\mathbf{1}\otimes\mathbf{1}B-\mathbf{1}\mathbf{A}\otimes \mathbf{B}\mathbf{1}= \mathbf{A}\otimes \mathbf{B} -\mathbf{A}\otimes \mathbf{B}=0
\end{align}
now we can use:
\begin{align}
    e^{\mathbf{A}\otimes \mathbf{1}+\mathbf{1}\otimes \mathbf{B}}=e^{\mathbf{X}+\mathbf{Y}}=e^\mathbf{X}e^\mathbf{Y}= e^{\mathbf{A}\otimes \mathbf{1}}e^{\mathbf{1}\otimes \mathbf{B}}
\end{align}
the general matrix exponential is 
\begin{align}
    e^\mathbf{X}&=\sum^\infty_{k=0}\frac{\mathbf{\mathbf{X}}^k}{k!}\\
    \Rightarrow     e^{\mathbf{A}\otimes \mathbf{1}}&=\sum^\infty_{k=0}\frac{(\mathbf{A}\otimes\mathbf{1})^k}{k!}
\end{align}
applying Eq.\ref{relation} repeatedly on $(\mathbf{A}\otimes\mathbf{1})^k$:
\begin{align}
(\mathbf{A}\otimes\mathbf{1})^k&=(\mathbf{A}\otimes\mathbf{1})(\mathbf{A}\otimes\mathbf{1})(\mathbf{A}\otimes\mathbf{1})^{k-2}=\\
    &=(\mathbf{A}^2\otimes\mathbf{1}^2)(\mathbf{A}\otimes\mathbf{1})(\mathbf{A}\otimes\mathbf{1})(\mathbf{A}\otimes\mathbf{1})^{k-4}\\
    &=(\mathbf{A}^2\otimes\mathbf{1}^2)(\mathbf{A}^2\otimes\mathbb{1^2})(\mathbf{A}\otimes\mathbf{1})^{k-4}\\
    &=(\mathbf{A}^4\otimes\mathbf{1}^4)(\mathbf{A}\otimes\mathbf{1})^{k-4}= ...\\
    &=\mathbf{A}^k\otimes\mathbf{1}^k
\end{align}
now we can write the exponential as:
\begin{align}
    e^{\mathbf{A}\otimes \mathbf{1}}&= \sum^\infty_{k=0}\frac{(\mathbf{A}\otimes\mathbf{1})^k}{k!}=\sum^\infty_{k=0}\frac{\mathbf{A}^k\otimes\mathbf{1}^k}{k!}\\
    \intertext{since $\mathbf{1}^k=\mathbf{1}$:}e^{\mathbf{A}\otimes \mathbf{1}}&=\sum^\infty_{k=0}\frac{\mathbf{A}^k}{k!}\otimes\mathbf{1}=e^\mathbf{A}\otimes\mathbf{1}
\end{align}
following the same arguments we see $e^{\mathbf{1}\otimes \mathbf{B}}=\mathbf{1}\otimes e^\mathbf{B}$ and thus we showed
\begin{align}
    e^{A\otimes \mathbf{1}+\mathbf{1}\otimes \mathbf{B}}&=e^{A\otimes \mathbf{1}}e^{\mathbf{1}\otimes \mathbf{B}}= (e^\mathbf{A}\otimes\mathbf{1})(\mathbf{1}\otimes e^\mathbf{B})\\
    &=(e^\mathbf{A}\mathbf{1})\otimes(\mathbf{1}e^\mathbf{B})=e^\mathbf{A}\otimes e^\mathbf{B}
\end{align}

\subsection*{(b)}
consider the state $\ket{a}\otimes\ket{b}$:
\begin{align}
    (\mathbf{A} \otimes \mathbf{B})(|a\rangle \otimes |b\rangle) &= (\mathbf{A}|a\rangle) \otimes (\mathbf{B}|b\rangle)\\
   \intertext{substitute given eigenvalues $\mathbf{A}|a\rangle = a|a\rangle$ and $\mathbf{B}|b\rangle = b|b\rangle$:} &=(a|a\rangle) \otimes (b|b\rangle) = ab(|a\rangle \otimes |b\rangle) 
\end{align}
So the eigenvalues are $ab$ with the eigenvectors $(|a\rangle \otimes |b\rangle)$.

\subsection*{(c)}
the standard Pauli Commutator is
\begin{align}
    [\sigma_i, \sigma_j] = 2i \epsilon_{ijk} \sigma_k
\end{align}
for the Tensor Product:
\begin{align}
    [\sigma_i \otimes \sigma_i, \sigma_j \otimes \sigma_j] &= (\sigma_i \otimes \sigma_i)(\sigma_j \otimes \sigma_j) - (\sigma_j \otimes \sigma_j)(\sigma_i \otimes \sigma_i)\\
    &=(\sigma_i \sigma_j \otimes \sigma_i \sigma_j) - (\sigma_j \sigma_i \otimes \sigma_j \sigma_i)
\end{align}
with $\sigma_i \sigma_j=\delta_{i,j}\mathbf{1}+i \epsilon_{ijk} \sigma_k$ we have to look at two cases.

case 1: $i=j \Rightarrow \sigma_i \sigma_j=\mathbf{1}$. The Tensor Product Commutator is then:
\begin{align}
    (\sigma_i \sigma_j \otimes \sigma_i \sigma_j) - (\sigma_j \sigma_i \otimes \sigma_j \sigma_i)&= (\mathbf{1}\otimes\mathbf{1})-(\mathbf{1}\otimes\mathbf{1})=0
\intertext{case 2: $i\neq j \Rightarrow \sigma_i \sigma_j=i \epsilon_{ijk} \sigma_k$. The Tensor Product Commutator is then:}
    (\sigma_i \sigma_j \otimes \sigma_i \sigma_j) - (\sigma_j \sigma_i \otimes \sigma_j \sigma_i)&= (i \sigma_k\otimes i\sigma_k)-(-i \sigma_k\otimes-i\sigma_k)\\
    &=-(\sigma_k\otimes\sigma_k)+(\sigma_k\otimes\sigma_k)= 0
\intertext{so for every $i,j$:}
    [\sigma_i \otimes \sigma_i, \sigma_j \otimes \sigma_j]&=0
\end{align}

\section*{Task 5.2: Creation-Annihilation Model}
\subsection*{(a)}
First column:\\
there are no transition from state $\emptyset\emptyset$ into another state $\rightarrow$ entire column is zero.\\

Second and third column:\\
$\mathcal{L}_{12,13}=0$: no transitions from $\emptyset A/ A\emptyset$ into state $\emptyset\emptyset$.\\
$\mathcal{L}_{23,32}=-D$: transition  $\emptyset A\leftrightarrow A\emptyset$ at rate D.\\
$\mathcal{L}_{24,34}=-\lambda$: transition $\emptyset A/ A\emptyset\rightarrow AA$ at rate $\lambda$.\\
$\mathcal{L}_{22,33}=(D+\lambda)$: has to be chosen such, that the column sum is zero, to ensure probability conservation.\\

Fourth column:\\
$\mathcal{L}_{14}=-\mu$: transition $AA\rightarrow \emptyset\emptyset$ at rate $\mu$.\\
$\mathcal{L}_{24,34}=0$: no transition from state $AA$ into states $\emptyset A/ A\emptyset$.\\
$\mathcal{L}_{44}=\mu$: has to be chosen such, that the column sum is zero, to ensure probability conservation.
\begin{align}
\mathcal{L}^{(2)}=
    \left[\begin{matrix}0 & 0 & 0 & - \mu\\0 & D + \lambda & - D & 0\\0 & - D & D + \lambda & 0\\0 & - \lambda & - \lambda & \mu\end{matrix}\right]
\end{align}

At $\lambda=0$ (no particle production), the ground state ($\epsilon=0$) is twofold degenerate. Because even and odd particle sectors cannot mix, they evolve independently into two distinct steady states, an even sector ($N=0, 2$): Particles fully annihilate, leaving the empty state:$$\ket{G_{\text{even}}} = \ket{\emptyset\emptyset\emptyset}$$
and an odd sector ($N=1, 3$): Particles reduce to a single diffusing particle, yielding a uniform distribution:$$\ket{G_{\text{odd}}} = \frac{1}{3}\big(\ket{\emptyset\emptyset A} + \ket{\emptyset A\emptyset} + \ket{A\emptyset\emptyset}\big)$$This independence prevents relaxation to a unique ground state, preserving the twofold degeneracy.
